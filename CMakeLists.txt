cmake_minimum_required(VERSION 3.20)
project(all-in-one-timer VERSION 1.0.0 LANGUAGES C)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "All In One Timer for OBS builds on Windows x64 with MSVC only (GDI text rendering).")
endif()

set(OBS_INSTALL_DIR "C:/Program Files/obs-studio" CACHE PATH "OBS Studio install folder (obs.dll is used to generate the import library)")
set(OBS_VERSION_TAG "32.0.1" CACHE STRING "obs-studio git tag the libobs headers come from (match your installed OBS)")
set(LIBOBS_SOURCE_DIR "${CMAKE_SOURCE_DIR}/deps/obs-studio" CACHE PATH "obs-studio checkout containing libobs/")

# Install straight into the OBS plugin folder by default
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  file(TO_CMAKE_PATH "$ENV{ProgramData}/obs-studio/plugins" _obs_plugins_dir)
  set(CMAKE_INSTALL_PREFIX "${_obs_plugins_dir}" CACHE PATH "OBS plugins folder" FORCE)
endif()

# --- libobs headers --------------------------------------------------------
# Only the headers are needed, so a sparse, shallow checkout of libobs/ is enough.
if(NOT EXISTS "${LIBOBS_SOURCE_DIR}/libobs/obs-module.h")
  find_package(Git REQUIRED)
  message(STATUS "Fetching libobs headers (obs-studio ${OBS_VERSION_TAG})")
  execute_process(
    COMMAND "${GIT_EXECUTABLE}" clone --quiet --depth 1 --branch ${OBS_VERSION_TAG} --filter=blob:none --sparse
            https://github.com/obsproject/obs-studio.git "${LIBOBS_SOURCE_DIR}"
    RESULT_VARIABLE _rc)
  if(NOT _rc EQUAL 0)
    message(FATAL_ERROR "git clone of obs-studio failed")
  endif()
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LIBOBS_SOURCE_DIR}" sparse-checkout set libobs RESULT_VARIABLE _rc)
  if(NOT _rc EQUAL 0)
    message(FATAL_ERROR "git sparse-checkout of libobs failed")
  endif()
endif()

# obsconfig.h is normally generated by the OBS build
set(_generated_dir "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${_generated_dir}/obsconfig.h" "#pragma once\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")

# --- obs.lib import library from the installed obs.dll ---------------------
set(_obs_dll "${OBS_INSTALL_DIR}/bin/64bit/obs.dll")
if(NOT EXISTS "${_obs_dll}")
  message(FATAL_ERROR "obs.dll not found at ${_obs_dll}. Set OBS_INSTALL_DIR to your OBS Studio folder.")
endif()

get_filename_component(_msvc_bin "${CMAKE_C_COMPILER}" DIRECTORY)
find_program(DUMPBIN_EXECUTABLE dumpbin HINTS "${_msvc_bin}" REQUIRED)
find_program(MSVC_LIB_EXECUTABLE lib HINTS "${_msvc_bin}" REQUIRED)

set(_obs_def "${CMAKE_BINARY_DIR}/obs.def")
set(_obs_lib "${CMAKE_BINARY_DIR}/obs.lib")

execute_process(COMMAND "${DUMPBIN_EXECUTABLE}" /nologo /exports "${_obs_dll}"
                OUTPUT_VARIABLE _exports RESULT_VARIABLE _rc)
if(NOT _rc EQUAL 0)
  message(FATAL_ERROR "dumpbin /exports failed on ${_obs_dll}")
endif()

# Export table lines look like: "    123   7A 0001A2B0 obs_source_create"
string(REGEX MATCHALL "\n[ \t]+[0-9]+[ \t]+[0-9A-Fa-f]+[ \t]+[0-9A-Fa-f]+[ \t]+[A-Za-z_][A-Za-z0-9_]*" _export_lines "${_exports}")
set(_def "LIBRARY obs\nEXPORTS\n")
foreach(_line IN LISTS _export_lines)
  string(REGEX REPLACE ".*[ \t]([A-Za-z_][A-Za-z0-9_]*)$" "\\1" _symbol "${_line}")
  string(APPEND _def "  ${_symbol}\n")
endforeach()
list(LENGTH _export_lines _num_exports)
if(_num_exports LESS 100)
  message(FATAL_ERROR "Could not read the export table of ${_obs_dll}")
endif()
file(WRITE "${_obs_def}" "${_def}")

execute_process(COMMAND "${MSVC_LIB_EXECUTABLE}" /nologo /machine:x64 "/def:${_obs_def}" "/out:${_obs_lib}"
                RESULT_VARIABLE _rc OUTPUT_QUIET)
if(NOT _rc EQUAL 0)
  message(FATAL_ERROR "Failed to generate obs.lib")
endif()
message(STATUS "Generated obs.lib from ${_obs_dll} (${_num_exports} exports)")

add_library(libobs SHARED IMPORTED)
set_target_properties(libobs PROPERTIES
  IMPORTED_IMPLIB "${_obs_lib}"
  IMPORTED_LOCATION "${_obs_dll}"
  INTERFACE_INCLUDE_DIRECTORIES "${LIBOBS_SOURCE_DIR}/libobs;${_generated_dir}")

# --- Plugin ----------------------------------------------------------------
add_library(all-in-one-timer MODULE src/all-in-one-timer.c)
set_target_properties(all-in-one-timer PROPERTIES PREFIX "" C_STANDARD 17 C_STANDARD_REQUIRED ON)
target_compile_definitions(all-in-one-timer PRIVATE _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE)
target_compile_options(all-in-one-timer PRIVATE /W3 /utf-8)
target_link_libraries(all-in-one-timer PRIVATE libobs gdi32 user32)

# Layout expected by OBS 28+:
#   <plugins>/all-in-one-timer/bin/64bit/all-in-one-timer.dll
#   <plugins>/all-in-one-timer/data/locale/*.ini
install(TARGETS all-in-one-timer LIBRARY DESTINATION all-in-one-timer/bin/64bit)
install(DIRECTORY data/ DESTINATION all-in-one-timer/data)
